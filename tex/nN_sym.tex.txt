\section{Geometry of the device}
The two end wires are fixed at $x_1=-0.5$ and $x_N=0.5$; the remaining wires are placed in the middle with equal spacing.
\begin{center}
\begin{tabular}{@{}c c l@{}}
\toprule
$N$ & $d$ & positions\\
\midrule
2 & 1 & $-0.5,\ 0.5$\\
3 & 0.5 & $-0.5,\ 0,\ 0.5$\\
4 & $1/3$ & $-0.5,\ -1/6,\ 1/6,\ 0.5$\\
5 & 0.25 & $-0.5,\ -0.25,\ 0,\ 0.25,\ 0.5$\\
\bottomrule
\end{tabular}
\end{center}
Spacing between wires: $\displaystyle d=\frac{1}{N-1}$.

\textbf{Meissner phase equation.} $\Phi_i=\Phi_1+2\pi b\,\Delta x_{1,i}$ with $\Delta x_{1,i}=(i-1)d$:
\begin{resbox}
\[
\Phi_i=\Phi_1+2\pi b\,(i-1)\,d,\qquad d=\frac{1}{N-1}
\]
\end{resbox}
\textbf{Testing it for $N=6$.} Positions $-0.5,-0.3,-0.1,0.1,0.3,0.5$ and $d=\frac{1}{N-1}=\frac{1}{6-1}=\frac15=0.2$:
\begin{align*}
\Phi_1&=\Phi_1+2\pi b(1-1)(0.2)=\Phi_1\\
\Phi_2&=\Phi_1+2\pi b(2-1)(0.2)=\Phi_1+0.4\pi b\\
\Phi_3&=\Phi_1+2\pi b(3-1)(0.2)=\Phi_1+0.8\pi b\\
\Phi_4&=\Phi_1+2\pi b(4-1)(0.2)=\Phi_1+1.2\pi b\\
\Phi_5&=\Phi_1+2\pi b(5-1)(0.2)=\Phi_1+1.6\pi b\\
\Phi_6&=\Phi_1+2\pi b(6-1)(0.2)=\Phi_1+2\pi b
\end{align*}
The last wire sees the full field phase $2\pi b$ relative to the first, as it must.

\section{Characteristic phase equation}
\[
\varphi=\frac{\Phi_1+\dots+\Phi_N}{N}=\frac{\sum_{i=1}^{N}\Phi_i}{N}=\frac{\Phi_1+\sum_{i=2}^{N}\Phi_i}{N}
\qquad\Longrightarrow\qquad N\varphi=\Phi_1+\sum_{i=2}^{N}\Phi_i .
\]
Using the Meissner phase equation $\Phi_i=\Phi_1+2\pi b(i-1)d$:
\begin{align*}
\Phi_1&=N\varphi-\sum_{i=2}^{N}\bigl(\Phi_1+2\pi b(i-1)d\bigr)=N\varphi-\underbrace{\sum_{i=2}^{N}\Phi_1}_{(N-1)\ \text{times}}-2\pi bd\sum_{i=2}^{N}(i-1)\\
\Phi_1&=N\varphi-\Phi_1(N-1)-2\pi bd\sum_{i=2}^{N}(i-1)=N\varphi-N\Phi_1+\Phi_1-2\pi bd\sum_{i=2}^{N}(i-1)\\
0&=N\varphi-N\Phi_1-2\pi bd\sum_{i=2}^{N}(i-1)
\quad\Longrightarrow\quad
\Phi_1=\varphi+\frac{2\pi bd\sum_{i=2}^{N}(i-1)}{-N}.
\end{align*}
\textbf{The summation.} With $j=i-1$,
\[
\sum_{i=2}^{N}(i-1)=1+2+3+\dots+(N-1)=\sum_{j=1}^{N-1}j=\frac{(N-1)(N-1+1)}{2}=\frac{(N-1)N}{2}.
\]
Therefore
\[
\Phi_1=\varphi+\frac{2\pi b\bigl(\frac{1}{N-1}\bigr)\frac{(N-1)N}{2}}{-N}=\varphi+\frac{\pi bN}{-N}
\qquad\Longrightarrow\qquad
\boxed{\Phi_1=\varphi-\pi b}
\]
\begin{notebox}{Note}
On the first try the notes have $\frac{N(N+1)}{2}$ for this sum; the correct value is $\frac{(N-1)N}{2}$ (the sum stops at $N-1$), which is what is used in the next line of the notes and here.
\end{notebox}

\textbf{Going back to the geometry of the device.} $x_1=-0.5$, so $x_i=-0.5+(i-1)d$ and $(i-1)d=x_i+0.5$. Substituting this back into the Meissner phase equation:
\[
\Phi_i=\Phi_1+2\pi b(i-1)d=\Phi_1+2\pi b(x_i+0.5)=\Phi_1+2\pi bx_i+\pi b ,
\]
and plugging in $\Phi_1=\varphi-\pi b$: $\Phi_i=\varphi-\pi b+2\pi bx_i+\pi b$, so
\begin{resbox}
\[
\Phi_i=\varphi+2\pi b\,x_i
\]
\end{resbox}

\section{Likharev CPR and current summation}
\[
\frac{I}{I_0}=\Bigl(\frac{\Phi}{\varphi_0}\Bigr)-\Bigl(\frac{\Phi}{\varphi_0}\Bigr)^3,\qquad
\frac{I}{I_0}=\sum_{i=1}^{N}\Bigl[\Bigl(\frac{\Phi_i}{\varphi_0}\Bigr)-\Bigl(\frac{\Phi_i}{\varphi_0}\Bigr)^3\Bigr]
\]
Substitute $\Phi_i=\varphi+2\pi bx_i$:
\begin{align*}
\frac{I}{I_0}&=\sum_{i=1}^{N}\Bigl[\frac{\varphi+2\pi bx_i}{\varphi_0}-\Bigl(\frac{\varphi+2\pi bx_i}{\varphi_0}\Bigr)^3\Bigr]
=\sum_{i=1}^{N}\Bigl[\frac{\varphi+2\pi bx_i}{\varphi_0}-\frac{\varphi^3+3\varphi^2(2\pi bx_i)+3\varphi(4\pi^2b^2x_i^2)+8\pi^3b^3x_i^3}{\varphi_0^3}\Bigr]\\
&=\sum_{i=1}^{N}\Bigl[\frac{\varphi_0^2(\varphi+2\pi bx_i)}{\varphi_0^3}-\frac{\varphi^3+6\varphi^2\pi bx_i+12\varphi\pi^2b^2x_i^2+8\pi^3b^3x_i^3}{\varphi_0^3}\Bigr]\\
&=\frac{1}{\varphi_0^3}\Bigl[\underbrace{\sum_{i=1}^{N}\varphi_0^2\varphi}_{(1)}+\underbrace{\sum_{i=1}^{N}2\pi b\varphi_0^2x_i}_{(2)}\\
&\qquad-\underbrace{\sum_{i=1}^{N}\varphi^3}_{(3)}-\underbrace{\sum_{i=1}^{N}6\varphi^2\pi bx_i}_{(4)}-\underbrace{\sum_{i=1}^{N}12\varphi\pi^2b^2x_i^2}_{(5)}-\underbrace{\sum_{i=1}^{N}8\pi^3b^3x_i^3}_{(6)}\Bigr]
\end{align*}
\begin{enumerate}[label=(\arabic*),leftmargin=*]
\item $\sum_{i=1}^{N}\varphi_0^2\varphi=N\varphi_0^2\varphi$.
\item $\sum_{i=1}^{N}2\pi b\varphi_0^2x_i=2\pi b\varphi_0^2\sum_{i=1}^{N}x_i=0$. Example: $\sum_{i=1}^{6}x_i=(-0.5)+(-0.3)+(-0.1)+0.1+0.3+0.5=0$.
\item $\sum_{i=1}^{N}\varphi^3=N\varphi^3$.
\item $\sum_{i=1}^{N}6\varphi^2\pi bx_i=6\varphi^2\pi b\sum_{i=1}^{N}x_i=0$ (same logic as (2)).
\item $\sum_{i=1}^{N}12\varphi\pi^2b^2x_i^2=12\varphi\pi^2b^2\sum_{i=1}^{N}x_i^2$: it is even, so it does not cancel.
\item $\sum_{i=1}^{N}8\pi^3b^3x_i^3=8\pi^3b^3\sum_{i=1}^{N}x_i^3=0$: it is odd, so it cancels.
\end{enumerate}
\[
\frac{I}{I_0}=\frac{1}{\varphi_0^3}\Bigl[N\varphi_0^2\varphi-N\varphi^3-12\varphi\pi^2b^2\sum_{i=1}^{N}x_i^2\Bigr]
\]

\section{The sum $\sum x_i^2$}
\begin{align*}
x_i&=-\frac12+(i-1)\cdot\frac{1}{N-1}=\frac{-(N-1)+2(i-1)}{2(N-1)}=\frac{1-N+2i-2}{2(N-1)}=\frac{2i-N-1}{2(N-1)}=\frac{2\bigl[i-\frac{N+1}{2}\bigr]}{2(N-1)}=\frac{i-\frac{N+1}{2}}{N-1}\\
x_i^2&=\frac{\bigl(i-\frac{N+1}{2}\bigr)^2}{(N-1)^2},\qquad
\sum_{i=1}^{N}x_i^2=\frac{1}{(N-1)^2}\sum_{i=1}^{N}\Bigl(i-\frac{N+1}{2}\Bigr)^2
\end{align*}
Expanding, $\bigl(i-\frac{N+1}{2}\bigr)^2=i^2-2i\frac{N+1}{2}+\bigl(\frac{N+1}{2}\bigr)^2=i^2-(N+1)i+\frac{(N+1)^2}{4}$, so
\[
\sum_{i=1}^{N}\Bigl(i-\frac{N+1}{2}\Bigr)^2=\underbrace{\sum_{i=1}^{N}i^2}_{(1)}-(N+1)\underbrace{\sum_{i=1}^{N}i}_{(2)}+\underbrace{\sum_{i=1}^{N}\frac{(N+1)^2}{4}}_{(3)}
\]
with (1) $\sum i^2=\frac{N(N+1)(2N+1)}{6}$, (2) $\sum i=\frac{N(N+1)}{2}$, (3) $\sum\frac{(N+1)^2}{4}=N\cdot\frac{(N+1)^2}{4}$:
\begin{align*}
\sum_{i=1}^{N}\Bigl(i-\frac{N+1}{2}\Bigr)^2&=\frac{N(N+1)(2N+1)}{6}-(N+1)\frac{N(N+1)}{2}+N\frac{(N+1)^2}{4}\\
&=N(N+1)\Bigl[\frac{2N+1}{6}-\frac{N+1}{2}+\frac{N+1}{4}\Bigr]=N(N+1)\Bigl(\frac{N-1}{12}\Bigr)
\end{align*}
\begin{resbox}
\[
\sum_{i=1}^{N}x_i^2=\frac{1}{(N-1)^2}\cdot\frac{N(N-1)(N+1)}{12}=\frac{N(N+1)}{12(N-1)}
\]
\end{resbox}
\begin{center}
\begin{tabular}{@{}c c c@{}}
\toprule
$N$ & direct $\sum x_i^2$ & $\frac{N(N+1)}{12(N-1)}$\\
\midrule
2 & $0.25+0.25=0.5$ & $\frac{6}{12}=0.5$\\
3 & $0.25+0+0.25=0.5$ & $\frac{12}{24}=0.5$\\
4 & $0.25+\frac1{36}+\frac1{36}+0.25=\frac59$ & $\frac{20}{36}=\frac59$\\
6 & $2(0.25+0.09+0.01)=0.7$ & $\frac{42}{60}=0.7$\\
\bottomrule
\end{tabular}
\end{center}

\section{Plugging back in}
\begin{align*}
\frac{I}{I_0}&=\frac{1}{\varphi_0^3}\Bigl[N\varphi_0^2\varphi-N\varphi^3-12\varphi\pi^2b^2\cdot\frac{N(N+1)}{12(N-1)}\Bigr]
=\frac{1}{\varphi_0^3}\Bigl[N\varphi_0^2\varphi-N\varphi^3-\frac{N(N+1)\varphi\pi^2b^2}{N-1}\Bigr]\\
&=\frac{1}{\varphi_0^3}\Bigl[N\varphi\Bigl(\varphi_0^2-\frac{(N+1)\pi^2b^2}{N-1}\Bigr)-N\varphi^3\Bigr]
=\frac{N}{\varphi_0^3}\Bigl[\Bigl(\varphi_0^2-\frac{(N+1)\pi^2b^2}{N-1}\Bigr)\varphi-\varphi^3\Bigr]
\end{align*}
\begin{resbox}
\[
\frac{I(\varphi)}{I_0}=\frac{N}{\varphi_0^3}\Bigl[\Bigl(\varphi_0^2-\frac{(N+1)\pi^2b^2}{N-1}\Bigr)\varphi-\varphi^3\Bigr]
\]
\end{resbox}
\begin{notebox}{A warning about factoring out $N$}
In two lines of the notes the bracket still reads $\varphi_0^2-\frac{N(N+1)\pi^2b^2}{N-1}$ after $N$ has already been factored out in front. The extra $N$ must go: the parity checks below only work with $\frac{(N+1)\pi^2b^2}{N-1}$. (With the extra $N$, $N=3$ would give $A=\frac{3\varphi_0^2-18\pi^2b^2}{\varphi_0^3}$, which matches neither the 3-wire result nor the Dayem paper at $N=2$.) The final highlighted boxes in the notes use the correct form.
\end{notebox}

\section{The Hamiltonian}
\[
I(\varphi)=\frac{N}{\varphi_0^3}\Bigl[\Bigl(\varphi_0^2-\frac{(N+1)\pi^2b^2}{N-1}\Bigr)\varphi-\varphi^3\Bigr]I_0,\qquad
A=\frac{N}{\varphi_0^3}\Bigl(\varphi_0^2-\frac{(N+1)\pi^2b^2}{N-1}\Bigr),\qquad D=\frac{N}{\varphi_0^3}.
\]
With $V=\frac{\hbar}{2e}\frac{d\varphi}{dt}$, $P=IV$:
\[
U(\varphi)=\int P\,dt=\int IV\,dt=\int I\frac{\hbar}{2e}\frac{d\varphi}{dt}dt=\frac{\hbar}{2e}\int_0^\varphi I(\varphi')\,d\varphi'=\frac{\hbar I_0}{2e}\Bigl(\frac A2\varphi^2-\frac D4\varphi^4\Bigr),
\]
and the kinetic term is the same as before, $-4E_C\,\partial_\varphi^2$.
\begin{resbox}
\textbf{Full Hamiltonian}
\[
\hat H=-4E_C\,\partial_\varphi^2+\frac{\hbar I_0}{2e}\Bigl[\frac{N}{2\varphi_0^3}\Bigl(\varphi_0^2-\frac{(N+1)\pi^2b^2}{N-1}\Bigr)\varphi^2-\frac{N}{4\varphi_0^3}\varphi^4\Bigr]
\]
\end{resbox}
\begin{paritybox}{: $N=2$ and $N=3$}
$N=2$: $A=\frac{2}{\varphi_0^3}(\varphi_0^2-3\pi^2b^2)=\frac{2\varphi_0^2-6\pi^2b^2}{\varphi_0^3}$, $D=\frac{2}{\varphi_0^3}$: the Dayem loop result. \checkmark\\
$N=3$: $A=\frac{3}{\varphi_0^3}(\varphi_0^2-2\pi^2b^2)=\frac{3\varphi_0^2-6\pi^2b^2}{\varphi_0^3}$, $D=\frac{3}{\varphi_0^3}$: the symmetric 3-nanowire result. \checkmark
\end{paritybox}
