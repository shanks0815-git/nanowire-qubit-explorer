\section{Realistic nanowire CPR models}
\[
\frac{I(\Phi)}{I_0}=\frac{\Phi}{\varphi_0}-a\Bigl(\frac{\Phi}{\varphi_0}\Bigr)^{2m+1}
\]
\begin{itemize}[leftmargin=*]
\item The CPR that was used before was valid at temperatures near the critical temperature, or for short nanowires.
\item Transmon qubits operate near absolute zero temperature; this is where the CPR can deviate severely from the cubic term.
\end{itemize}

\Delta x_{12}=x_2-(-0.5)=x_2+0.5,\qquad \Delta x_{23}=0.5-x_2,\qquad \Delta x_{13}=x_3-x_1=1 .

\section{Meissner phase equations and characteristic phase}
\begin{align*}
\Phi_2&=\Phi_1+2\pi b\,\Delta x_{12}\\
\Phi_3&=\Phi_2+2\pi b\,\Delta x_{23}=\Phi_1+2\pi b\,\Delta x_{12}+2\pi b\,\Delta x_{23}=\Phi_1+2\pi b(\Delta x_{12}+\Delta x_{23})=\Phi_1+2\pi b
\end{align*}
\[
\varphi=\frac{\Phi_1+\Phi_2+\Phi_3}{3}=\frac{3\Phi_1+2\pi b\Delta x_{12}+2\pi b}{3}
\quad\Longrightarrow\quad
\boxed{\Phi_1=\varphi-\frac{2\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}
\]
(this is $\Phi_1$ in terms of the characteristic phase).

\section{The CPR as a summation}
Using the current sum rule and the new CPR:
\begin{align*}
\frac{I}{I_0}&=\Bigl[\frac{\Phi_1}{\varphi_0}-a\Bigl(\frac{\Phi_1}{\varphi_0}\Bigr)^{2m+1}\Bigr]+\Bigl[\frac{\Phi_2}{\varphi_0}-a\Bigl(\frac{\Phi_2}{\varphi_0}\Bigr)^{2m+1}\Bigr]+\Bigl[\frac{\Phi_3}{\varphi_0}-a\Bigl(\frac{\Phi_3}{\varphi_0}\Bigr)^{2m+1}\Bigr]\\
&=\frac{\Phi_1+\Phi_1+\Phi_1+2\pi b\Delta x_{12}+2\pi b}{\varphi_0}-a\Bigl[\Bigl(\frac{\Phi_1}{\varphi_0}\Bigr)^{2m+1}+\Bigl(\frac{\Phi_1+2\pi b\Delta x_{12}}{\varphi_0}\Bigr)^{2m+1}+\Bigl(\frac{\Phi_1+2\pi b}{\varphi_0}\Bigr)^{2m+1}\Bigr].
\end{align*}
Using $\Phi_1=\varphi-\frac{2\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}$, the linear part is
$\frac{3\Phi_1+2\pi b\Delta x_{12}+2\pi b}{\varphi_0}=\frac{3\varphi}{\varphi_0}$, and
\begin{align*}
\frac{I}{I_0}=\frac{3\varphi}{\varphi_0}-a\Biggl[&\underbrace{\Bigl(\frac{\varphi-\frac{2\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}{\varphi_0}\Bigr)^{2m+1}}_{\textcircled{1}}\\
&+\underbrace{\Bigl(\frac{\varphi+\frac{4\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}{\varphi_0}\Bigr)^{2m+1}}_{\textcircled{2}}+\underbrace{\Bigl(\frac{\varphi-\frac{2\pi b\Delta x_{12}}{3}+\frac{4\pi b}{3}}{\varphi_0}\Bigr)^{2m+1}}_{\textcircled{3}}\Biggr]
\end{align*}

\section{Binomial theorem}
$(a+b)^n=\sum_{k=0}^{n}\binom nk a^{n-k}b^k$, $\binom nk=\frac{n!}{k!(n-k)!}$. We name the three shifts
\[
z=\frac{-2\pi b\Delta x_{12}-2\pi b}{3},\qquad
w=\frac{4\pi b\Delta x_{12}-2\pi b}{3},\qquad
t=\frac{-2\pi b\Delta x_{12}+4\pi b}{3},
\]
so that \textcircled{1} $=\bigl(\frac{\varphi+z}{\varphi_0}\bigr)^{2m+1}$, \textcircled{2} $=\bigl(\frac{\varphi+w}{\varphi_0}\bigr)^{2m+1}$, \textcircled{3} $=\bigl(\frac{\varphi+t}{\varphi_0}\bigr)^{2m+1}$, and
\begin{gather*}
(\varphi+z)^{2m+1}=\sum_{k=0}^{2m+1}\binom{2m+1}{k}\varphi^{2m+1-k}z^k,\qquad
(\varphi+w)^{2m+1}=\sum_{k=0}^{2m+1}\binom{2m+1}{k}\varphi^{2m+1-k}w^k,\\
(\varphi+t)^{2m+1}=\sum_{k=0}^{2m+1}\binom{2m+1}{k}\varphi^{2m+1-k}t^k .
\end{gather*}
Now we combine all three terms:
\[
\sum_{k=0}^{2m+1}\binom{2m+1}{k}\varphi^{2m+1-k}\bigl(z^k+w^k+t^k\bigr).
\]
\textbf{$k=0$:} $\binom{2m+1}{0}=\frac{(2m+1)!}{0!\,(2m+1)!}=1$, so the term is $\varphi^{2m+1}(z^0+w^0+t^0)=3\varphi^{2m+1}$.

\textbf{$k=1$:} $\binom{2m+1}{1}=\frac{(2m+1)!}{1!\,(2m)!}=2m+1$, so the term is $(2m+1)\varphi^{2m}(z+w+t)$, and
\[
z+w+t=\frac{-2\pi b\Delta x_{12}-2\pi b+4\pi b\Delta x_{12}-2\pi b-2\pi b\Delta x_{12}+4\pi b}{3}=0 .
\]
Combining with the $k=0$ term and the rest of the expansion:
\begin{resbox}
\[
(\varphi+z)^{2m+1}+(\varphi+w)^{2m+1}+(\varphi+t)^{2m+1}=3\varphi^{2m+1}+\sum_{k=2}^{2m+1}\binom{2m+1}{k}\varphi^{2m+1-k}\bigl(z^k+w^k+t^k\bigr)
\]
\end{resbox}

\section{Newton's identities (Newton--Girard formula)}
Define the power sums and the elementary symmetric polynomials of $z,w,t$:
\[
M_k=z^k+w^k+t^k,\qquad e_1=z+w+t,\quad e_2=zw+zt+wt,\quad e_3=zwt .
\]
Newton's identities with $e_1=0$ give
\[
M_2=-2e_2,\qquad M_3=3e_3,\qquad M_k=-e_2M_{k-2}+e_3M_{k-3}\quad(k\ge4).
\]
\textbf{$e_1$.} Shown above: $e_1=0$.

\textbf{$e_2$.}
\begin{align*}
e_2&=\Bigl(\tfrac{-2\pi b\Delta x_{12}-2\pi b}{3}\Bigr)\Bigl(\tfrac{4\pi b\Delta x_{12}-2\pi b}{3}\Bigr)+\Bigl(\tfrac{-2\pi b\Delta x_{12}-2\pi b}{3}\Bigr)\Bigl(\tfrac{-2\pi b\Delta x_{12}+4\pi b}{3}\Bigr)+\Bigl(\tfrac{4\pi b\Delta x_{12}-2\pi b}{3}\Bigr)\Bigl(\tfrac{-2\pi b\Delta x_{12}+4\pi b}{3}\Bigr)\\
&=\frac{-8\pi^2b^2\Delta x_{12}^2+4\pi^2b^2-4\pi^2b^2\Delta x_{12}}{9}+\frac{4\pi^2b^2\Delta x_{12}^2-8\pi^2b^2-4\pi^2b^2\Delta x_{12}}{9}+\frac{-8\pi^2b^2\Delta x_{12}^2+20\pi^2b^2\Delta x_{12}-8\pi^2b^2}{9}\\
&=\frac{-12\pi^2b^2\Delta x_{12}^2+12\pi^2b^2\Delta x_{12}-12\pi^2b^2}{9}=-\frac{4\pi^2b^2}{3}\bigl(\Delta x_{12}^2-\Delta x_{12}+1\bigr)
\end{align*}
\textbf{$e_3$.}
\begin{align*}
e_3&=\Bigl(\tfrac{-2\pi b\Delta x_{12}-2\pi b}{3}\Bigr)\Bigl(\tfrac{4\pi b\Delta x_{12}-2\pi b}{3}\Bigr)\Bigl(\tfrac{-2\pi b\Delta x_{12}+4\pi b}{3}\Bigr)\\
&=\frac{16\pi^3b^3\Delta x_{12}^3-32\pi^3b^3\Delta x_{12}^2-8\pi^3b^3\Delta x_{12}+16\pi^3b^3+8\pi^3b^3\Delta x_{12}^2-16\pi^3b^3\Delta x_{12}}{27}\\
&=\frac{16\pi^3b^3\Delta x_{12}^3-24\pi^3b^3\Delta x_{12}^2-24\pi^3b^3\Delta x_{12}+16\pi^3b^3}{27}
=\frac{8\pi^3b^3\bigl(2\Delta x_{12}^3-3\Delta x_{12}^2-3\Delta x_{12}+2\bigr)}{27}\\
&=\frac{8\pi^3b^3(\Delta x_{12}-2)(2\Delta x_{12}^2+\Delta x_{12}-1)}{27}
=\frac{8\pi^3b^3(\Delta x_{12}-2)(\Delta x_{12}+1)(2\Delta x_{12}-1)}{27}
\end{align*}
Therefore
\begin{align*}
M_2&=-2e_2=-2\cdot\Bigl(-\frac{4\pi^2b^2}{3}\Bigr)(\Delta x_{12}^2-\Delta x_{12}+1)=\frac{8\pi^2b^2}{3}\bigl(\Delta x_{12}^2-\Delta x_{12}+1\bigr),\\
M_3&=3e_3=\frac{8\pi^3b^3}{9}(\Delta x_{12}-2)(\Delta x_{12}+1)(2\Delta x_{12}-1),\\
M_k&=-e_2M_{k-2}+e_3M_{k-3}\qquad(k\ge4).
\end{align*}
Plugging this into the current--phase formula that we had originally:
\begin{resbox}
\[
\frac{I(\varphi)}{I_0}=\frac{3\varphi}{\varphi_0}-\frac{a}{\varphi_0^{2m+1}}\Bigl[3\varphi^{2m+1}+\sum_{k=2}^{2m+1}\binom{2m+1}{k}\varphi^{2m+1-k}M_k\Bigr]
\]
where $M_2=-2e_2$, $M_3=3e_3$, $M_k=-e_2M_{k-2}+e_3M_{k-3}$ ($k\ge4$).
\end{resbox}

\section{The Hamiltonian}
\textbf{Kinetic term.} $T=\frac{Q^2}{2C}$, $Q=2e\hat N$, $T=\frac{4e^2\hat N^2}{2C}$, $[\hat\varphi,\hat N]=i$, $\hat N=-i\partial_\varphi$, $E_C=\frac{e^2}{2C}$:\quad $\boxed{T=-4E_C\,\partial_\varphi^2}$.

\textbf{Potential term.} $V=\frac{\hbar}{2e}\frac{d\varphi}{dt}$, $P=IV$, $U(\varphi)=\int P\,dt=\frac{\hbar}{2e}\int_0^\varphi I(\varphi')\,d\varphi'$. Now we integrate the CPR in terms of $\varphi$:
\begin{align*}
U(\varphi)&=\frac{\hbar I_0}{2e}\int_0^{\varphi}\Bigl[\frac{3\varphi'}{\varphi_0}-\frac{a}{\varphi_0^{2m+1}}\Bigl(3\varphi'^{2m+1}+\sum_{k=2}^{2m+1}\binom{2m+1}{k}\varphi'^{2m+1-k}M_k\Bigr)\Bigr]d\varphi'
\end{align*}
Splitting up the integration:
\begin{align*}
U(\varphi)&=\frac{\hbar I_0}{2e}\Bigl[\int_0^{\varphi}\frac{3\varphi'}{\varphi_0}d\varphi'-\frac{a}{\varphi_0^{2m+1}}\int_0^{\varphi}3\varphi'^{2m+1}d\varphi'-\frac{a}{\varphi_0^{2m+1}}\sum_{k=2}^{2m+1}\binom{2m+1}{k}M_k\int_0^{\varphi}\varphi'^{2m+1-k}d\varphi'\Bigr]\\
&=\frac{\hbar I_0}{2e}\Bigl[\frac{3\varphi^2}{2\varphi_0}-\frac{a}{\varphi_0^{2m+1}}\frac{3\varphi^{2m+2}}{2m+2}-\frac{a}{\varphi_0^{2m+1}}\sum_{k=2}^{2m+1}\binom{2m+1}{k}M_k\frac{\varphi^{2m+2-k}}{2m+2-k}\Bigr]
\end{align*}
\begin{resbox}
\[
\hat H=T+U=-4E_C\,\partial_\varphi^2+\frac{\hbar I_0}{2e}\Bigl[\frac{3\varphi^2}{2\varphi_0}-\frac{a}{\varphi_0^{2m+1}}\frac{3\varphi^{2m+2}}{2m+2}-\frac{a}{\varphi_0^{2m+1}}\sum_{k=2}^{2m+1}\binom{2m+1}{k}M_k\,\frac{\varphi^{2m+2-k}}{2m+2-k}\Bigr]
\]
\end{resbox}

\begin{paritybox}{: $m=1$, $a=1$ (Likharev) and $\Delta x_{12}=0.5$ (symmetric)}
Remember that the Likharev CPR is $\frac{I}{I_0}=\frac{\varphi}{\varphi_0}-\bigl(\frac{\varphi}{\varphi_0}\bigr)^3$; the realistic CPR with $m=1$, $a=1$ gives exactly this. With $m=1$, $a=1$:
\[
\hat H=-4E_C\,\partial_\varphi^2+\frac{\hbar I_0}{2e}\Bigl[\frac{3\varphi^2}{2\varphi_0}-\frac{1}{\varphi_0^3}\frac{3\varphi^4}{4}-\frac{1}{\varphi_0^3}\sum_{k=2}^{3}\binom3kM_k\frac{\varphi^{4-k}}{4-k}\Bigr].
\]
$k=2$: $\binom32=\frac{3!}{2!\,1!}=3$, so the term is $3M_2\frac{\varphi^2}{2}=4\pi^2b^2\varphi^2(\Delta x_{12}^2-\Delta x_{12}+1)$.\\
$k=3$: $\binom33=1$, so the term is $M_3\varphi=\frac{8\pi^3b^3}{9}\varphi(\Delta x_{12}-2)(\Delta x_{12}+1)(2\Delta x_{12}-1)$.
Plugging back into the Hamiltonian,
\[
U(\varphi)=\frac{\hbar I_0}{2e}\Bigl[\frac{3\varphi^2}{2\varphi_0}-\frac{3\varphi^4}{4\varphi_0^3}-\frac{4\pi^2b^2\varphi^2}{\varphi_0^3}(\Delta x_{12}^2-\Delta x_{12}+1)-\frac{8\pi^3b^3}{9\varphi_0^3}\varphi(\Delta x_{12}-2)(\Delta x_{12}+1)(2\Delta x_{12}-1)\Bigr],
\]
which is the asymmetric Likharev result. For the symmetric case $\Delta x_{12}=0.5$: $\Delta x_{12}^2-\Delta x_{12}+1=0.75$ and $(-1.5)(1.5)(0)=0$, so
\begin{align*}
U(\varphi)&=\frac{\hbar I_0}{2e}\Bigl[\frac{3\varphi^2}{2\varphi_0}-\frac{3\varphi^4}{4\varphi_0^3}-\frac{3\pi^2b^2\varphi^2}{\varphi_0^3}\Bigr]
=\frac{\hbar I_0}{2e}\Bigl[\varphi^2\Bigl(\frac{3\varphi_0^2}{2\varphi_0^3}-\frac{3\pi^2b^2}{\varphi_0^3}\Bigr)-\frac{3\varphi^4}{4\varphi_0^3}\Bigr]\\
&=\frac{\hbar I_0}{2e}\Bigl[\varphi^2\,\frac{3\varphi_0^2-6\pi^2b^2}{2\varphi_0^3}-\frac{3\varphi^4}{4\varphi_0^3}\Bigr]
=\frac{\hbar I_0}{2e}\Bigl[\frac12A\varphi^2-\frac14D\varphi^4\Bigr],\quad A=\frac{3\varphi_0^2-6\pi^2b^2}{\varphi_0^3},\ D=\frac{3}{\varphi_0^3}.
\end{align*}
Choosing $m=1$, $a=1$ confirms that the general $2m+1$ CPR collapses back to the Likharev CPR; choosing $\Delta x_{12}=0.5$ confirms that the asymmetric derivation collapses back to the symmetric one. \checkmark
\end{paritybox}
