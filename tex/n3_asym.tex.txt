\textbf{Introduction.} Now we are introducing the same thing except the wire $x_2$ is not in its central position, $x_2\neq0$. We are going to write the Hamiltonian for this case and produce the potential energy function as a function of $x_2$. ($x_2\neq0$ also breaks the parity symmetry)
\section{Geometry of the device}
\begin{figure}[H]\centering
\device{0/x_1=-0.5,3.9/x_2,6/x_3=0.5}{\draw[dashed,thin] (3,0)--(3,3.0); \node[above] at (3,3.4){\small $x=0$}; \node at (0,3.75){$\Phi_1$}; \node at (3.9,3.75){$\Phi_2$}; \node at (6,3.75){$\Phi_3$};}
\caption{The middle wire is moved from the central position $x=0$ (dashed) to $x_2$.}
\end{figure}
\[
\Delta x_{12}=x_2-(-0.5)=x_2+0.5,\qquad \Delta x_{23}=0.5-x_2,\qquad \Delta x_{13}=x_3-x_1=1 .
\]

\section{Meissner phase equations}
\begin{align*}
\Phi_2&=\Phi_1+2\pi b\,\Delta x_{12}\\
\Phi_3&=\Phi_2+2\pi b\,\Delta x_{23}=\Phi_1+2\pi b\,\Delta x_{12}+2\pi b\,\Delta x_{23}=\Phi_1+2\pi b(\Delta x_{12}+\Delta x_{23})=\Phi_1+2\pi b
\end{align*}

\section{Characteristic phase}
\begin{align*}
\varphi&=\frac{\Phi_1+\Phi_2+\Phi_3}{3}=\frac{\Phi_1+\Phi_1+2\pi b\Delta x_{12}+\Phi_1+2\pi b}{3}=\frac{3\Phi_1+2\pi b\Delta x_{12}+2\pi b}{3}\\
3\varphi-2\pi b\Delta x_{12}-2\pi b&=3\Phi_1\qquad\Longrightarrow\qquad
\boxed{\Phi_1=\varphi-\frac{2\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}
\end{align*}
This is $\Phi_1$ in terms of the characteristic phase. Then
\[
\Phi_2=\varphi+\frac{4\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3},\qquad
\Phi_3=\varphi-\frac{2\pi b\Delta x_{12}}{3}+\frac{4\pi b}{3}.
\]

\section{Likharev CPR in terms of the characteristic phase}
\[
\frac{I}{I_0}=\Bigl[\frac{\Phi_1}{\varphi_0}-\Bigl(\frac{\Phi_1}{\varphi_0}\Bigr)^3\Bigr]+\Bigl[\frac{\Phi_2}{\varphi_0}-\Bigl(\frac{\Phi_2}{\varphi_0}\Bigr)^3\Bigr]+\Bigl[\frac{\Phi_3}{\varphi_0}-\Bigl(\frac{\Phi_3}{\varphi_0}\Bigr)^3\Bigr]
\]
Rewriting in terms of the characteristic phase, the three brackets are \textcircled{1}, \textcircled{2}, \textcircled{3}:
\begin{align*}
\frac{I}{I_0}=\ &\underbrace{\Bigl[\frac{\varphi-\frac{2\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}{\varphi_0}-\Bigl(\frac{\varphi-\frac{2\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}{\varphi_0}\Bigr)^3\Bigr]}_{\textcircled{1}}
+\underbrace{\Bigl[\frac{\varphi+\frac{4\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}{\varphi_0}-\Bigl(\frac{\varphi+\frac{4\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}}{\varphi_0}\Bigr)^3\Bigr]}_{\textcircled{2}}\\
&+\underbrace{\Bigl[\frac{\varphi-\frac{2\pi b\Delta x_{12}}{3}+\frac{4\pi b}{3}}{\varphi_0}-\Bigl(\frac{\varphi-\frac{2\pi b\Delta x_{12}}{3}+\frac{4\pi b}{3}}{\varphi_0}\Bigr)^3\Bigr]}_{\textcircled{3}}
\end{align*}

\subsection*{Cube \textcircled{1}}
\begin{align*}
\Bigl(\varphi-\tfrac{2\pi b\Delta x_{12}}{3}-\tfrac{2\pi b}{3}\Bigr)^3=\ &\varphi^3-2\pi b(\Delta x_{12}+1)\varphi^2+\frac{4\pi^2b^2}{3}(\Delta x_{12}^2+2\Delta x_{12}+1)\varphi\\
&-\frac{8\pi^3b^3}{27}\Delta x_{12}^3-\frac{8\pi^3b^3}{9}\Delta x_{12}^2-\frac{8\pi^3b^3}{9}\Delta x_{12}-\frac{8\pi^3b^3}{27}
\end{align*}
\subsection*{Cube \textcircled{2}}
\begin{align*}
\Bigl(\varphi+\tfrac{4\pi b\Delta x_{12}}{3}-\tfrac{2\pi b}{3}\Bigr)^3=\ &\varphi^3+(4\pi b\Delta x_{12}-2\pi b)\varphi^2+\Bigl(\frac{16\pi^2b^2\Delta x_{12}^2}{3}-\frac{16\pi^2b^2\Delta x_{12}}{3}+\frac{4\pi^2b^2}{3}\Bigr)\varphi\\
&+\frac{64\pi^3b^3\Delta x_{12}^3}{27}-\frac{32\pi^3b^3\Delta x_{12}^2}{9}+\frac{16\pi^3b^3\Delta x_{12}}{9}-\frac{8\pi^3b^3}{27}
\end{align*}
\subsection*{Cube \textcircled{3}}
\begin{align*}
\Bigl(\varphi-\tfrac{2\pi b\Delta x_{12}}{3}+\tfrac{4\pi b}{3}\Bigr)^3=\ &\varphi^3+(4\pi b-2\pi b\Delta x_{12})\varphi^2+\Bigl(\frac{4\pi^2b^2\Delta x_{12}^2}{3}-\frac{16\pi^2b^2\Delta x_{12}}{3}+\frac{16\pi^2b^2}{3}\Bigr)\varphi\\
&-\frac{8\pi^3b^3\Delta x_{12}^3}{27}+\frac{16\pi^3b^3\Delta x_{12}^2}{9}-\frac{32\pi^3b^3\Delta x_{12}}{9}+\frac{64\pi^3b^3}{27}
\end{align*}

\subsection*{Adding \textcircled{1}, \textcircled{2}, \textcircled{3}}
Linear parts: $\frac{1}{\varphi_0}\bigl(\varphi-\frac{2\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}+\varphi+\frac{4\pi b\Delta x_{12}}{3}-\frac{2\pi b}{3}+\varphi+\frac{4\pi b}{3}-\frac{2\pi b\Delta x_{12}}{3}\bigr)=\frac{3\varphi}{\varphi_0}$.

Cubic parts, power by power:
\begin{align*}
\varphi^3&:\ 1+1+1=3\\
\varphi^2&:\ -2\pi b\Delta x_{12}-2\pi b+4\pi b\Delta x_{12}-2\pi b+4\pi b-2\pi b\Delta x_{12}=0\\
\varphi^1&:\ \frac{4\pi^2b^2}{3}\bigl[(\Delta x_{12}^2+2\Delta x_{12}+1)+(4\Delta x_{12}^2-4\Delta x_{12}+1)+(\Delta x_{12}^2-4\Delta x_{12}+4)\bigr]=8\pi^2b^2(\Delta x_{12}^2-\Delta x_{12}+1)\\
\varphi^0&:\ \frac{8\pi^3b^3}{27}\bigl[(-1+8-1)\Delta x_{12}^3\bigr]+\frac{8\pi^3b^3}{9}\bigl[(-1-4+2)\Delta x_{12}^2+(-1+2-4)\Delta x_{12}\bigr]+\frac{8\pi^3b^3}{27}(-1-1+8)\\
&\ \ =\frac{8\pi^3b^3}{9}\bigl(2\Delta x_{12}^3-3\Delta x_{12}^2-3\Delta x_{12}+2\bigr)
\end{align*}
\begin{resbox}
\[
\frac{I}{I_0}=\frac{1}{\varphi_0}(3\varphi)-\frac{1}{\varphi_0^3}\Bigl[3\varphi^3+8\pi^2b^2(\Delta x_{12}^2-\Delta x_{12}+1)\varphi+\frac{8\pi^3b^3}{9}\bigl(2\Delta x_{12}^3-3\Delta x_{12}^2-3\Delta x_{12}+2\bigr)\Bigr]
\]
\end{resbox}
\begin{paritybox}{: $x_2=0$}
Plug in $x_2=0$, i.e.\ $\Delta x_{12}=0.5$: $\Delta x_{12}^2-\Delta x_{12}+1=\tfrac34$ and $2(\tfrac18)-3(\tfrac14)-3(\tfrac12)+2=0$, so
\[
\frac{I}{I_0}\Big|_{x_2=0}=\frac{3\varphi}{\varphi_0}-\frac{3\varphi^3}{\varphi_0^3}-\frac{6\pi^2b^2\varphi}{\varphi_0^3},
\]
the same as the symmetric case, so the formula checks out. \checkmark
\end{paritybox}

\section{Writing the CPR in terms of $A$, $D$ and $R$}
Format: $\frac{I}{I_0}=A\varphi-D\varphi^3-R$.
\begin{align*}
\frac{I}{I_0}&=\frac{3\varphi}{\varphi_0}-\frac{3\varphi^3}{\varphi_0^3}-\frac{8\pi^2b^2\Delta x_{12}^2\varphi}{\varphi_0^3}+\frac{8\pi^2b^2\Delta x_{12}\varphi}{\varphi_0^3}-\frac{8\pi^2b^2\varphi}{\varphi_0^3}-\frac{8\pi^3b^3}{9\varphi_0^3}\bigl(2\Delta x_{12}^3-3\Delta x_{12}^2-3\Delta x_{12}+2\bigr)\\
&=\underbrace{\Bigl(\frac{3\varphi_0^2-8\pi^2b^2(\Delta x_{12}^2-\Delta x_{12}+1)}{\varphi_0^3}\Bigr)}_{A}\varphi-\underbrace{\Bigl(\frac{3}{\varphi_0^3}\Bigr)}_{D}\varphi^3-\underbrace{\frac{8\pi^3b^3}{9\varphi_0^3}\bigl(2\Delta x_{12}^3-3\Delta x_{12}^2-3\Delta x_{12}+2\bigr)}_{R}
\end{align*}
\textbf{In terms of just $x_2$.} We know $\Delta x_{12}=x_2-x_1=x_2+0.5$, and
\[
(x_2+0.5)^2=x_2^2+x_2+0.25,\qquad (x_2+0.5)^3=x_2^3+1.5x_2^2+0.75x_2+0.125 .
\]
Then $\Delta x_{12}^2-\Delta x_{12}+1=x_2^2+\tfrac34$ and
$2\Delta x_{12}^3-3\Delta x_{12}^2-3\Delta x_{12}+2=2x_2^3-\tfrac92x_2=\tfrac12x_2(4x_2^2-9)$, so
\begin{resbox}
\[
\frac{I}{I_0}=\Bigl(\frac{3\varphi_0^2-8\pi^2b^2x_2^2-6\pi^2b^2}{\varphi_0^3}\Bigr)\varphi-\Bigl(\frac{3}{\varphi_0^3}\Bigr)\varphi^3-\frac{4\pi^3b^3x_2(4x_2^2-9)}{9\varphi_0^3}
\]
\[
\frac{I}{I_0}=A\varphi-D\varphi^3-R,\qquad I(\varphi)=(A\varphi-D\varphi^3-R)I_0
\]
\[
A=\frac{3\varphi_0^2-8\pi^2b^2x_2^2-6\pi^2b^2}{\varphi_0^3},\qquad D=\frac{3}{\varphi_0^3},\qquad R=\frac{4\pi^3b^3x_2(4x_2^2-9)}{9\varphi_0^3}
\]
\end{resbox}
\begin{notebox}{Remark}
The cubic in $\Delta x_{12}$ factors as $2\Delta x_{12}^3-3\Delta x_{12}^2-3\Delta x_{12}+2=(2\Delta x_{12}-1)(\Delta x_{12}-2)(\Delta x_{12}+1)$, so $R$ vanishes exactly at $\Delta x_{12}=\tfrac12$ ($x_2=0$): the tilt is a pure parity-breaking term.
\end{notebox}

\section{Integrating to get the potential}
With $V=\frac{\hbar}{2e}\frac{d\varphi}{dt}$ and $P=IV$,
\begin{align*}
U(\varphi)=\int P\,dt=\int IV\,dt=\int I\Bigl(\frac{\hbar}{2e}\frac{d\varphi}{dt}\Bigr)dt=\frac{\hbar}{2e}\int_0^{\varphi}I(\varphi')\,d\varphi'
=\frac{\hbar I_0}{2e}\int_0^{\varphi}(A\varphi'-D\varphi'^3-R)\,d\varphi'
\end{align*}
\[
U(\varphi)=\frac{\hbar I_0}{2e}\Bigl(\frac{A\varphi^2}{2}-\frac{D\varphi^4}{4}-R\varphi\Bigr).
\]
We know the kinetic energy term of the Hamiltonian is $-4E_C\,\partial_\varphi^2$.

\section{Final Hamiltonian}
\begin{resbox}
\[
\hat H=-4E_C\,\partial_\varphi^2+\frac{\hbar I_0}{2e}\Bigl(\frac12A\varphi^2-\frac14D\varphi^4-R\varphi\Bigr)
\]
\[
A=\frac{3\varphi_0^2-8\pi^2b^2x_2^2-6\pi^2b^2}{\varphi_0^3},\qquad D=\frac{3}{\varphi_0^3},\qquad R=\frac{4\pi^3b^3x_2(4x_2^2-9)}{9\varphi_0^3}
\]
\end{resbox}
